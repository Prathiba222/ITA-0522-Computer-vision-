import cv2
import numpy as np
import matplotlib.pyplot as plt

# Create a white image
img = np.ones((300, 300), dtype=np.uint8) * 255

# Draw a large black rectangle
cv2.rectangle(img, (50, 50), (250, 250), 0, -1)

# Add small black objects
cv2.circle(img, (30, 30), 5, 0, -1)
cv2.circle(img, (270, 270), 5, 0, -1)

# Create kernel
kernel = np.ones((21, 21), np.uint8)

# Apply Black Hat operation
blackhat = cv2.morphologyEx(img, cv2.MORPH_BLACKHAT, kernel)

# Display images
plt.figure(figsize=(10, 4))

plt.subplot(1, 2, 1)
plt.imshow(img, cmap='gray')
plt.title("Original Image")
plt.axis("off")

plt.subplot(1, 2, 2)
plt.imshow(blackhat, cmap='gray')
plt.title("Black Hat Operation")
plt.axis("off")

plt.show()
