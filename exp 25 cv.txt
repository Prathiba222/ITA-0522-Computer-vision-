import cv2
import numpy as np
import matplotlib.pyplot as plt
from google.colab import files

# Upload reference watch image
print("Upload a reference watch image:")
uploaded = files.upload()
reference_path = list(uploaded.keys())[0]

# Upload test image
print("Upload an image to recognize:")
uploaded = files.upload()
test_path = list(uploaded.keys())[0]

# Read images
reference = cv2.imread(reference_path)
test = cv2.imread(test_path)

# Convert images to grayscale
gray1 = cv2.cvtColor(reference, cv2.COLOR_BGR2GRAY)
gray2 = cv2.cvtColor(test, cv2.COLOR_BGR2GRAY)

# Create ORB detector
orb = cv2.ORB_create(nfeatures=1000)

# Detect keypoints and descriptors
kp1, des1 = orb.detectAndCompute(gray1, None)
kp2, des2 = orb.detectAndCompute(gray2, None)

# Match features
if des1 is not None and des2 is not None:
    bf = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=True)
    matches = bf.match(des1, des2)

    # Sort matches
    matches = sorted(matches, key=lambda x: x.distance)

    # Draw matching features
    result = cv2.drawMatches(
        reference, kp1,
        test, kp2,
        matches[:30], None,
        flags=2
    )

    # Display result
    plt.figure(figsize=(14, 6))
    plt.imshow(cv2.cvtColor(result, cv2.COLOR_BGR2RGB))
    plt.title("Watch Recognition Using OpenCV")
    plt.axis("off")
    plt.show()

    # Recognition result
    if len(matches) >= 10:
        print("Watch recognized in the given image!")
    else:
        print("Watch not recognized.")
else:
    print("Unable to detect features. Try different images.")
